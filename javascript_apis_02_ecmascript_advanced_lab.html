function lab0Reduce() {
  const arr = [6, 8, 10];

  // reduce((累積値, 現在値) => 次の累積値, 初期値)
  const sum = arr.reduce((acc, x) => acc + x, 0);
  // 0 + 6 = 6 → 6 + 8 = 14 → 14 + 10 = 24

  console.log(`元: [${arr}]`);
  console.log(`合計: ${sum}`);  // 24
  console.log(`平均: ${sum / arr.length}`);  // 8
  return sum;
}

Set — 一意値の集合

✓ iOS ✓ Android

重複を許さない集合。add/has/deleteがすべてO(1)。配列の重複排除に最適。

💡 利用シーン

  3. タグ管理(追加/削除/存在確認が頻繁)
配列で重複チェック: includes() を毎回呼ぶ → O(n)

Setを使えば: has() / add() がO(1)

JS — 「速度比較(1万件)」ボタンの中身
// 「速度比較」ボタン: 1万件のhas判定でArray vs Setを計測
function lab2Perf() {
  const N = 10000;
  const arr = Array.from({length:N}, (_,i) => i);
  const set = new Set(arr);

  // Array.includes() を1万回
  const t1 = performance.now();
  for (let i = 0; i < N; i++) arr.includes(i);
  const arrTime = performance.now() - t1;

  // Set.has() を1万回
  const t2 = performance.now();
  for (let i = 0; i < N; i++) set.has(i);
  const setTime = performance.now() - t2;

  console.log(`Array.includes: ${arrTime.toFixed(2)}ms`);
  console.log(`Set.has:        ${setTime.toFixed(2)}ms`);
  console.log(`Setが ${(arrTime/setTime).toFixed(0)}倍高速`);
}

WeakMap/WeakRef — 弱参照

✓ iOS ✓ Android

キーへの弱参照。元オブジェクトが他から参照されなくなれば自動的にGCで消える。メモリリーク防止に有効。

💡 利用シーン
  1. DOM要素に紐づくメタデータの保管(要素削除時に自動破棄)
  2. 大きなオブジェクトの一時キャッシュ(GCを妨げない)
  3. プライベートフィールドの実装(オブジェクトに紐づける)
📐 「弱参照」とは?
普通の参照(Map等)はオブジェクトを「掴んだまま」=GCの対象外にする。
弱参照(WeakMap等)は「軽く触れているだけ」=他に参照がなくなればGCで消える。
これにより「メモリ解放を意識しなくていい」キャッシュが作れる。 
通常のMap:
  Map: { obj => "data" }
       └─ 強参照 ──→ obj
  obj への外部参照を全て消しても、Mapが掴み続ける → メモリに残る

WeakMap:
  WeakMap: { obj 〜→ "data" }
            └─ 弱参照 ─→ obj
  obj への外部参照を消すと → GC対象 → WeakMapから自動消去

// 「reviver」ボタン: パース時に値を変換
function lab4Reviver() {
  const jsonStr = '{"name":"Tom","createdAt":"2026-05-06T12:00:00Z"}';

  // reviver: 各キー値ペアに対して変換処理
  const obj = JSON.parse(jsonStr, (key, value) => {
    // ISO日付文字列なら Date オブジェクトに変換
    if (typeof value === 'string' &&
        /^\d{4}-\d{2}-\d{2}T/.test(value)) {
      return new Date(value);
    }
    return value;
  });

  console.log(`name: ${obj.name}`);  // 文字列のまま
  console.log(`createdAt: ${obj.createdAt}`);
  console.log(`is Date: ${obj.createdAt instanceof Date}`);  // true
  return obj;
}

// 「allSettled」ボタン: 失敗込みで全結果回収
async function lab6Settled() {
  const results = await Promise.allSettled([
    delay(500, 'A'),
    delay(300, 'B失敗', true),
    delay(700, 'C')
  ]);

  // 各要素は { status: 'fulfilled'|'rejected', value/reason: ... }
  results.forEach((r, i) => {
    if (r.status === 'fulfilled') {
      console.log(`[${i}] OK: ${r.value}`);
    } else {
      console.log(`[${i}] NG: ${r.reason}`);
    }
  });
}

✓ iOS ✓ Android

// 「独自Range作成」ボタン: Symbol.iteratorで反復可能化
function lab9Range() {
  const range = {
    from: 1,
    to: 5,
    // Symbol.iterator: for...ofが内部で呼ぶメソッド
    [Symbol.iterator]() {
      let current = this.from;
      const last = this.to;
      return {
        // next(): 次の値を {value, done} で返す
        next() {
          if (current <= last) {
            return { value: current++, done: false };
          } else {
            return { value: undefined, done: true };
          }
        }
      };
    }
  };

  for (const n of range) {
    console.log(`値: ${n}`);
  }
}
JS — 「スプレッド対応」ボタンの中身
// 「スプレッド対応」ボタン: Iteratorは...でも展開可能
function lab9Spread() {
  const range = {
    from: 10, to: 14,
    *[Symbol.iterator]() {  // Generatorで簡潔に書ける
      for (let i = this.from; i <= this.to; i++) yield i;
    }
  };

  const arr = [...range];   // Iteratorからスプレッドで配列化
  console.log(`スプレッド: [${arr}]`);

  const max = Math.max(...range);  // 関数引数にも展開
  console.log(`max: ${max}`);

  const [first, ...rest] = range;  // 分割代入も
  console.log(`first: ${first}, rest: [${rest}]`);
}
