Function — bind/call/apply

✓ iOS ✓ Android

関数のthisを固定するbind、即時呼出のcall/apply、引数数を取るfn.length。

💡 利用シーン
  1. イベントハンドラのthisをコンポーネント自身に固定(bind)
  2. 配列の最大値を求める(Math.max.apply(null, arr))
  3. クラス外から配列メソッドを借用([].slice.call(arguments))
┌────────┬──────────────┬────────────┬──────────────────┐
│ メソッド │ 実行タイミング │ 引数の渡し方 │ 戻り値           │
├────────┼──────────────┼────────────┼──────────────────┤
│ bind   │ 後で実行     │ 個別に列挙   │ 新しい関数(再利用) │
│ call   │ その場で実行  │ 個別に列挙   │ 関数の戻り値      │
│ apply  │ その場で実行  │ 配列で渡す   │ 関数の戻り値      │
└────────┴──────────────┴────────────┴──────────────────┘
共通点: 第1引数で this を指定する

fn.bind(obj, a, b)  → 後で fn() を呼ぶ用の新関数を作る
fn.call(obj, a, b)  → 今すぐ fn(a,b) を obj.fn(a,b) として呼ぶ
fn.apply(obj, [a,b])→ 今すぐ fn(a,b) を obj.fn(a,b) として呼ぶ(配列形式)
📌 「Math.max.apply(null, [...])」のnullは何?
apply の第1引数は本来「this の中身」を指定する場所。でも Math.max は内部で this を使わない関数なので、何を渡しても結果は同じ。「使わないよ」を明示する慣例として null を渡す(undefined や {} でも動く)。
逆に user.greet.apply(user, [...]) のように this を使うメソッドでは、第1引数で適切なオブジェクトを必ず指定する必要がある。

// 「④apply」ボタン: その場で実行(引数を配列で渡す)
function lab1Apply() {
  const args = ['Hey', '!?'];   // 引数を配列に

  // apply(this指定, 引数の配列) → 即実行
  const result = lab1User.greet.apply(lab1User, args);
  console.log(`apply結果: ${result}`);   // "Hey, Tom!?"

  // ━━━ 典型用途: 配列を関数の個別引数に展開 ━━━
  const nums = [3, 1, 4, 1, 5, 9];

  // Math.max は引数を個別に列挙する関数:
  //   Math.max(3, 1, 4, 1, 5, 9) → OK
  //   Math.max([3,1,4,1,5,9])    → NG (配列はNaNになる)
  // → applyを使うと配列を個別引数に展開できる

  // 第1引数の null の意味:
  //   apply(this指定, 引数配列) の「this指定」は本来必須だが、
  //   Math.max は内部で this を使わない関数なので何でも良い。
  //   「使わないよ」を明示する慣例として null を渡す。
  //   undefined や {} でも動くが、null が業界標準。
  const max = Math.max.apply(null, nums);  // 9
  console.log(`Math.max.apply: ${max}`);

  // ━━━ 現代では ... スプレッド構文が同じ用途で使える ━━━
  // Math.max(...nums)  → 9  (ES2015以降。これが推奨)
  // applyは古い環境(IE等)互換が必要なときの選択肢
}

function lab4Throw(kind) {
  try {
    if (kind === 'type')  throw new TypeError('値の型が違います');
    if (kind === 'range') throw new RangeError('範囲外の値です');
    if (kind === 'cause') {
      try { JSON.parse('{bad json'); }
      catch(orig) {
        // causeで原因チェーンを保持
        throw new Error('処理失敗', { cause: orig });
      }
    }
  } catch(e) {
    console.log(`name: ${e.name}`);
    console.log(`message: ${e.message}`);
    console.log(`instanceof Error: ${e instanceof Error}`);
    if (e.cause) {
      console.log(`cause: ${e.cause.name}: ${e.cause.message}`);
    }
  }
}

Number — 数値判定とEPSILON

✓ iOS ✓ Android

Number.isFinite/isInteger/isNaNで型安全な判定。EPSILONで浮動小数点比較。

💡 利用シーン
  1. フォーム入力値の妥当性チェック(NaNや無限大を除外)
  2. 金額計算で0.1+0.2 !== 0.3問題をEPSILONで回避
  3. DB主キーが整数かチェック
📐 EPSILON(イプシロン)とは?
ギリシャ文字の ε(エプシロン/イプシロン)。数学では「ごく小さな正の数」を表す慣習的な記号として使われる。

Number.EPSILON は、JavaScriptが扱える最小の「1との差」=約 2.22 × 10⁻¹⁶(0.0000000000000002...)。
これは「これより小さい違いはコンピュータでは表現できない」という計算の精度限界を示す値。

浮動小数点演算の誤差(下記の0.1+0.2問題)はだいたいこの精度限界くらいで発生するため、「2つの値の差がEPSILON未満なら実質同じ」と判定するために使う。 
なぜ 0.1 + 0.2 ≠ 0.3 なのか?
─────────────────────────────
コンピュータは2進数で数値を保持する。
0.1 や 0.2 は2進数では循環小数(無限に続く)になる。
0.1(10進) → 0.0001100110011001100...(2進、無限)

→ どこかで打ち切るしかない → 微小な誤差が残る
→ 演算結果が 0.30000000000000004 になる

0.1 + 0.2          → 0.30000000000000004  ← 浮動小数点誤差!
0.1 + 0.2 === 0.3  → false                ← 直接比較は危険

解決法: 差がEPSILON以下なら「実質同じ」とみなす
Math.abs(a - b) < Number.EPSILON
        ↓
    差の絶対値 が 微小値ε より小さければ等価扱い

// 「0.1+0.2 を比較」ボタンが実行する処理
function lab6Run() {
  const a = 0.1 + 0.2;

  // 直接比較は失敗する
  const direct = (a === 0.3);
  console.log(`0.1 + 0.2 = ${a}`);                  // 0.30000000000000004
  console.log(`a === 0.3 → ${direct}`);              // false

  // EPSILON で誤差吸収
  const epsilon = Math.abs(a - 0.3) < Number.EPSILON;
  console.log(`Number.EPSILON = ${Number.EPSILON}`);
  console.log(`|a - 0.3| < EPSILON → ${epsilon}`);     // true

  // その他のNumber判定メソッド
  Number.isFinite(42);     // true
  Number.isFinite("42");   // false (型変換しない)
  Number.isInteger(3.0);   // true (3と等価)
}

// 「2^53超の値を比較」ボタンが実行する処理
function lab7Run() {
  const target = "9007199254740993";  // 2^53 + 1

  // Number は精度欠損
  const asNum = Number(target);
  console.log(`Number("${target}") → ${asNum}`);
  console.log(`欠損: ${asNum.toString() !== target}`);  // true

  // BigInt は精度保持
  const asBig = BigInt(target);
  console.log(`BigInt("${target}") → ${asBig}n`);
  console.log(`欠損: ${asBig.toString() !== target}`);  // false

  // BigInt演算: 同士のみOK
  const doubled = asBig * 2n;  // OK
  // asBig * 2  → TypeError (BigIntとNumberは混合不可)

  console.log(`MAX_SAFE_INTEGER: ${Number.MAX_SAFE_INTEGER}`);
  return { asNum, asBig, doubled };
}

// 「名前付きキャプチャ実行」ボタンが実行する処理
function lab12Run() {
  const v = document.getElementById('lab12-input').value;

  // (?<名前>...) で名前付きキャプチャグループ
  const re = /(?<year>\d{4})-(?<month>\d{2})-(?<day>\d{2})/;
  const m = v.match(re);

  if (m) {
    console.log(`年: ${m.groups.year}`);   // "2026"
    console.log(`月: ${m.groups.month}`);  // "05"
    console.log(`日: ${m.groups.day}`);    // "06"
    return m.groups;
  } else {
    console.error('YYYY-MM-DD形式で入力してください');
  }
}
